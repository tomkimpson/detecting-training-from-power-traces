% --- nodes -------------------------------------------------------------------

	{\textbf{Dominant run}\\ training ($\mathcal{H}_1$) or\\ inference ($\mathcal{H}_0$)};
\node[blk=2.6cm, fill=okorange!12] (bg) at (1.3, -0.72)
	{\textbf{Background}\\ small trainings,\\ fine-tunings};
\node[sum] (s1) at (3.3, 0) {$+$};
\node[blk=2.5cm, fill=okorange!12, minimum height=0.6cm] (p0) at (3.3, -1.85)
	{static overhead $P_0$};

	{\textbf{Delivery path} $h_{\mathrm{del}}$\\ PSU, PDU, storage\\ (low-pass, notch)};

	{\textbf{Meter} $h_{\mathrm{ins}}$, $f_s$\\ integrate, calibrate,\\ sample};
\node[sum] (s2) at (15.7, 0) {$+$};
\node[blk=3.2cm, fill=okblue!10, minimum height=0.6cm] (eta) at (15.7, -1.85)

	{\textbf{Verifier}\\ detector\\ (\cref{sec:methods})};

% --- waveform sketches (one period = 0.43 units) -------------------------------
% (1) sharp two-phase cadence leaving the facility
\begin{scope}[shift={(3.8, 0.16)}]
	\draw[sk] (0,0.36)--(0.26,0.36)--(0.26,0.08)--(0.43,0.08)--(0.43,0.36)

		--(1.12,0.08)--(1.29,0.08);
\end{scope}

\begin{scope}[shift={(8.75, 0.16)}]
	\draw[sk, domain=0:1.29, samples=60, smooth]
		plot (\x, {0.22 + 0.09*sin(deg(2*pi*\x/0.43))});
\end{scope}
% (3) integrated and sampled by the meter
\begin{scope}[shift={(13.95, 0.16)}]
